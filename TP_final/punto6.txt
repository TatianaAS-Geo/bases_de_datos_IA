6. Top 3 países productores de películas

db.movies.aggregate([
  { $unwind: "$countries" },
  {
    $group: {
      _id: "$countries",
      total: { $sum: 1 }
    }
  },
  { $sort: { total: -1 } },
  { $limit: 3 }
])

{
  _id: 'USA',
  total: 10921
}

{
  _id: 'UK',
  total: 2652
}

{
  _id: 'France',
  total: 2647
}
