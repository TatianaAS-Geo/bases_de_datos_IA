2. Calificación promedio de IMDb por género

db.movies.aggregate([
  { $unwind: "$genres" },
  {
    $group: {
      _id: "$genres",
      avgRating: { $avg: "$imdb.rating" }
    }
  },
  { $sort: { avgRating: -1 } }
])
