4. Número total de películas y duración promedio por director
db.movies.aggregate([
  {
    $group: {
      _id: "$directors",
      totalPeliculas: { $sum: 1 },
      avgDuracion: { $avg: "$runtime" }
    }
  },
  { $sort: { totalPeliculas: -1 } }
])
