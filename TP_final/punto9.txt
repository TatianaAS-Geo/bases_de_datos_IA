9. Número total de películas comentadas por usuario

db.comments.aggregate([
  {
    $group: {
      _id: "$email",
      peliculasComentadas: { $addToSet: "$movie_id" }
    }
  },
  {
    $project: {
      totalPeliculas: { $size: "$peliculasComentadas" }
    }
  },
  { $sort: { totalPeliculas: -1 } }
])
