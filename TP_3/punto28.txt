28. Verificar empresas comunes entre dos personas.

MATCH (a:Persona {apellido: "Corrado"})-[:TRABAJO]->(e:Empresa)<-[:TRABAJO]-(b:Persona)
WHERE a <> b
RETURN b.nombre, b.apellido, e.nombre AS empresa
