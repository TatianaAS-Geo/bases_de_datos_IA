4. Personas que estudiaron carreras no universitarias y los nombres de las carreras.

MATCH (p:Persona)-[:ESTUDIO]->(c:Carrera)
WHERE c.nivel <> "Universitario"
RETURN p.nombre, p.apellido, c.nombre
